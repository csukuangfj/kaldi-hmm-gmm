75inline float LogAdd(float x, float y) {
76 float diff;
77
78 if (x < y) {
79 diff = x - y;
80 x = y;
81 } else {
82 diff = y - x;
83 }
84 // diff is negative. x is now the larger one.
85
86 if (diff >= kMinLogDiffFloat) {
87 float res;
88 res = x + Log1p(std::exp(diff));
89 return res;
90 } else {
91 return x; // return the larger one.
92 }

117static inline bool ApproxEqual(float a, float b,
118 float relative_tolerance = 0.001) {
119 // a==b handles infinities.
120 if (a == b) return true;
121 float diff = std::abs(a - b);
122 if (diff == std::numeric_limits<float>::infinity() || diff != diff)
123 return false; // diff is +inf or nan.
124 return (diff <= relative_tolerance * (std::abs(a) + std::abs(b)));
125}

142template <class I>
143I Gcd(I m, I n) {
144 if (m == 0 || n == 0) {
145 if (m == 0 && n == 0) { // gcd not defined, as all integers are divisors.
146 KHG_ERR << "Undefined GCD since m = 0, n = 0.";
147 }
148 return (m == 0 ? (n > 0 ? n : -n) : (m > 0 ? m : -m));
149 // return absolute value of whichever is nonzero
150 }
151 // could use compile-time assertion
152 // but involves messing with complex template stuff.
153 KHG_ASSERT(std::numeric_limits<I>::is_integer);
154 while (1) {
155 m %= n;
156 if (m == 0) return (n > 0 ? n : -n);
157 n %= m;
158 if (n == 0) return (m > 0 ? m : -m);
159 }
160}
