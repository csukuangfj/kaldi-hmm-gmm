34template <typename T>
35inline bool IsSorted(const std::vector<T> &vec) {
36 typename std::vector<T>::const_iterator iter = vec.begin(), end = vec.end();
37 if (iter == end) return true;
38 while (1) {
39 typename std::vector<T>::const_iterator next_iter = iter;
40 ++next_iter;
41 if (next_iter == end) return true; // end of loop and nothing out of order
42 if (*next_iter < *iter) return false;
43 iter = next_iter;
44 }
45}

71template <class T>
72inline void WriteIntegerVector(std::ostream &os, bool binary,
73 const std::vector<T> &v) {
74 // Compile time assertion that this is not called with a wrong type.
75 static_assert(std::is_integral<T>::value, "");
76 if (binary) {
77 char sz = sizeof(T); // this is currently just a check.
78 os.write(&sz, 1);
79 int32_t vecsz = static_cast<int32_t>(v.size());
80 KHG_ASSERT((size_t)vecsz == v.size());
81
82 os.write(reinterpret_cast<const char *>(&vecsz), sizeof(vecsz));
83 if (vecsz != 0) {
84 os.write(reinterpret_cast<const char *>(&(v[0])), sizeof(T) * vecsz);
85 }
86 } else {
87 // focus here is on prettiness of text form rather than
88 // efficiency of reading-in.
89 // reading-in is dominated by low-level operations anyway:
90 // for efficiency use binary.
91 os << "[ ";
92 typename std::vector<T>::const_iterator iter = v.begin(), end = v.end();
93 for (; iter != end; ++iter) {
94 if (sizeof(T) == 1)
95 os << static_cast<int16_t>(*iter) << " ";
96 else
97 os << *iter << " ";
98 }
99 os << "]\n";
100 }
101 if (os.fail()) {
102 KHG_ERR << "Write failure in WriteIntegerVector.";
103 }
104}

108template <class T>
109inline void ReadIntegerVector(std::istream &is, bool binary,
110 std::vector<T> *v) {
111 static_assert(std::is_integral<T>::value, "");
112 KHG_ASSERT(v != nullptr);
113 if (binary) {
114 int sz = is.peek();
115 if (sz == sizeof(T)) {
116 is.get();
117 } else { // this is currently just a check.
118 KHG_ERR << "ReadIntegerVector: expected to see type of size " << sizeof(T)
119 << ", saw instead " << sz << ", at file position " << is.tellg();
120 }
121 int32_t vecsz;
122 is.read(reinterpret_cast<char *>(&vecsz), sizeof(vecsz));
123 if (is.fail() || vecsz < 0) goto bad;
124
125 v->resize(vecsz);
126
127 if (vecsz > 0) {
128 is.read(reinterpret_cast<char *>(&((*v)[0])), sizeof(T) * vecsz);
129 }
130 } else {
131 std::vector<T> tmp_v; // use temporary so v doesn't use extra memory
132 // due to resizing.
133 is >> std::ws;
134 if (is.peek() != static_cast<int>('[')) {
135 KHG_ERR << "ReadIntegerVector: expected to see [, saw " << is.peek()
136 << ", at file position " << is.tellg();
137 }
138 is.get(); // consume the '['.
139 is >> std::ws; // consume whitespace.
140 while (is.peek() != static_cast<int>(']')) {
141 if (sizeof(T) == 1) { // read/write chars as numbers.
142 int16_t next_t;
143 is >> next_t >> std::ws;
144 if (is.fail())
145 goto bad;
146 else
147 tmp_v.push_back((T)next_t);
148 } else {
149 T next_t;
150 is >> next_t >> std::ws;
151 if (is.fail())
152 goto bad;
153 else
154 tmp_v.push_back(next_t);
155 }
156 }
157 is.get(); // get the final ']'.
158 *v = tmp_v; // could use std::swap to use less temporary memory, but this
159 // uses less permanent memory.
160 }
161 if (!is.fail()) return;
162bad:
163 KHG_ERR << "ReadIntegerVector: read failure at file position " << is.tellg();
164}

168template <class A>
169void DeletePointers(std::vector<A *> *v) {
170 KHG_ASSERT(v != nullptr);
171 typename std::vector<A *>::iterator iter = v->begin(), end = v->end();
172 for (; iter != end; ++iter) {
173 if (*iter != nullptr) {
174 delete *iter;
175 *iter = nullptr; // set to NULL for extra safety.
176 }
177 }
178}

181template <typename Int1, typename Int2 = Int1>
182struct PairHasher { // hashing function for pair<int>
183 size_t operator()(const std::pair<Int1, Int2> &x) const noexcept {
184 // 7853 was chosen at random from a list of primes.
185 return x.first + x.second * 7853;
186 }
187 PairHasher() { // Check we're instantiated with an integer type.
188 static_assert(std::is_integral<Int1>::value, "");
189 static_assert(std::is_integral<Int2>::value, "");
190 }
191};

194template <class A>
195bool ContainsNullPointers(const std::vector<A *> &v) {
196 typename std::vector<A *>::const_iterator iter = v.begin(), end = v.end();
197 for (; iter != end; ++iter)
198 if (*iter == static_cast<A *>(nullptr)) return true;
199 return false;

203template <typename Int>
204struct VectorHasher { // hashing function for vector<Int>.
205 size_t operator()(const std::vector<Int> &x) const noexcept {
206 size_t ans = 0;
207 auto iter = x.begin(), end = x.end();
208 for (; iter != end; ++iter) {
209 ans *= kPrime;
210 ans += *iter;
211 }
212 return ans;
213 }
214 VectorHasher() { // Check we're instantiated with an integer type.
215 static_assert(std::is_integral<Int>::value, "");
216 }
217
218 private:
219 static const int kPrime = 7853;
220};
