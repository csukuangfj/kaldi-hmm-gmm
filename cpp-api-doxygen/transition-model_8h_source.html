439 Tuple(int32_t phone, int32_t hmm_state, int32_t forward_pdf,
440 int32_t self_loop_pdf)
441 : phone(phone),
442 hmm_state(hmm_state),
443 forward_pdf(forward_pdf),
444 self_loop_pdf(self_loop_pdf) {}
445
449 bool operator<(const Tuple &other) const {
450 if (phone < other.phone) {
451 return true;
452 } else if (phone > other.phone) {
453 return false;
454 } else if (hmm_state < other.hmm_state) {
455 return true;
456 } else if (hmm_state > other.hmm_state) {
457 return false;
458 } else if (forward_pdf < other.forward_pdf) {
459 return true;
460 } else if (forward_pdf > other.forward_pdf) {
461 return false;
462 } else {
463 return (self_loop_pdf < other.self_loop_pdf);
464 }
465 }
466
467 bool operator==(const Tuple &other) const {
468 return (phone == other.phone && hmm_state == other.hmm_state &&
469 forward_pdf == other.forward_pdf &&
470 self_loop_pdf == other.self_loop_pdf);
471 }
