409 bool Map(const EventType &event, EventAnswerType *ans) const override {
410 EventValueType tmp;
411
412 *ans = -1; // means no answer
413
414 if (Lookup(event, key_, &tmp) && tmp >= 0 &&
415 tmp < (EventValueType)table_.size() && table_[tmp] != nullptr) {
416 return table_[tmp]->Map(event, ans);
417 }
418 return false;
419 }
420
421 void GetChildren(std::vector<EventMap *> *out) const override {
422 out->clear();
423 for (size_t i = 0; i < table_.size(); ++i) {
424 if (table_[i] != nullptr) {
425 out->push_back(table_[i]);
426 }
427 }
428 }
429
430 void MultiMap(const EventType &event,
431 std::vector<EventAnswerType> *ans) const override {
432 EventValueType tmp;
433 if (Lookup(event, key_, &tmp)) {
434 if (tmp >= 0 && tmp < (EventValueType)table_.size() &&
435 table_[tmp] != nullptr)
436 return table_[tmp]->MultiMap(event, ans);
437 // else no answers.
438 } else { // all answers are possible if no such key.
439 for (size_t i = 0; i < table_.size(); ++i) {
440 if (table_[i] != nullptr) {
441 // append.
442 table_[i]->MultiMap(event, ans);
443 }
444 }
445 }
446 }
